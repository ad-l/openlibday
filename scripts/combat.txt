# Combat: lock a target, move into range and shoot (projectile and impact).
# Run: openlibday --headless --script scripts/combat.txt  (needs LIBDAY_ROOT)
skirmish Opportunity Knocks
key Return
key Return
key Return
deploy Grenadier
tile 44 18
deploy Beast Wagon
tile 44 20
key Return
key Y
# Enemies are never selected (the first unit is selected at the start).
tile 35 19
expect selected 0
# The Beast Wagon cannot hit the Footman at (35, 19) from here: the lock
# proposes a path to a tile it can shoot from.
tile 44 20
expect selected 28
tile 35 19
expect selected 28
expect lock.target Unit(9)
expect lock.pending false
expect preview.target.0 35
expect preview.target.1 25
# Confirm: it walks there, faces the target and shoots (no projectile).
tile 35 19
expect lock.pending true
idle 2000
expect lock null
expect shot null
expect units.-1.x 35
expect units.-1.y 25
expect units.-1.facing 4
expect units.-1.ap 75
# Animation sounds: the walk (one per facing), the shot, the explosion.
expect sounds.0 BLCA\P1\BLCA1M0.SMK
expect sounds.-2 BLCA\P1\BLCA1T4.SMK
expect sounds.-1 BFOT\P0\BFOT0E0.SMK
# The Footman is destroyed: it explodes, leaves its wreck and the unit
# list closes up behind it.
expect ruins.0.0 UBRUI141
expect ruins.0.1 35
expect ruins.0.2 19
expect units.-1.index 27
expect units.9.name Footman
expect units.9.armor 10
# The Grenadier lobs a grenade at the Footman at (35, 15): two turns'
# walk, then it shoots from in range.
tile 44 18
tile 35 15
tile 35 15
idle 2000
button EndTurn
idle 2000
tile 42 16
tile 35 15
tile 35 15
idle 2000
button EndTurn
idle 2000
# Seed 5 rolls a miss: the grenade scatters to a neighbouring tile.
seed 5
tile 40 14
tile 35 15
expect preview.target.0 39
tile 35 15
wait 30
expect shot.unit 26
expect shot.path.0.0 1988
expect shot.aim.0 35
expect shot.aim.1 15
expect shot.lands.0 36
expect shot.lands.1 14
expect shot.damage 10
idle 2000
# The blast (DamageArea 3) reaches the Footman one tile away at half
# strength: 10 - 10 / 2.
expect units.9.armor 5
expect units.-2.x 39
expect units.-2.ap 20
expect cursor CINVA
# Next turn. FIRE arms a single shot at anything, bare ground included.
button EndTurn
idle 3000
tile 35 25
button Fire
expect fire_mode true
hover 36 22
expect cursor CTARG
tile 36 22
expect lock.target Tile(36, 22)
expect lock.once true
tile 36 22
expect fire_mode false
wait 5
expect shot.target Tile(36, 22)
expect shot.repeat false
idle 3000
expect shot null
expect units.27.ap 195
# Lock and fire keeps shooting while it can: the Pack Freak takes two
# shots and is destroyed.
tile 30 26
tile 30 26
wait 5
expect shot.repeat true

expect shot.end 0
wait 28

expect shot.repeat false
idle 8000
expect shot null

expect units.-1.name Beast Wagon
expect units.-1.ap 85
